  <script>
    const totalSteps = 5;

    function updateProgress(step) {
      const percent = (step / totalSteps) * 100;
      document.getElementById('progressFill').style.width = percent + '%';
    }

    function nextSession(step) {
      document.querySelectorAll('[id^="session"], #intro').forEach(e => e.classList.add('hidden'));
      document.getElementById(`session${step}`).classList.remove('hidden');
      updateProgress(step);
    }

    function showFinish() {
      document.querySelectorAll('[id^="session"], #intro').forEach(e => e.classList.add('hidden'));
      document.getElementById('finish-screen').classList.remove('hidden');

      // Fortschritt speichern
      localStorage.setItem(`day${DAY_NUMBER}Completed`, 'true');

      // Unlock-Time für nächsten Tag (06:00 Uhr lokal)
      const now = new Date();
      const unlock = new Date(now);
      unlock.setDate(now.getDate() + 1);
      unlock.setHours(6, 0, 0, 0);
      localStorage.setItem(`day${DAY_NUMBER + 1}UnlockTime`, unlock.getTime());

      // Erfolgssound abspielen
      const sound = document.getElementById("successSound");
      if (sound) {
        sound.volume = 0.5;
        sound.play().catch(() => {});
      }
    }

    document.getElementById("dayLabel").textContent = DAY_NUMBER;
  </script>
